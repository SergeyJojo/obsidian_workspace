Вот реализация метода `Run`, Сергей, которая решает твою задачу:

- собирает сессии по `ProjectID` из нескольких каналов (`chs`)
    
- отправляет агрегированные данные:
    
    - либо по таймеру `FlushInterval`
        
    - либо при достижении `MaxSessionsPerProject` по конкретному проекту
        

---

### ✅ Реализация `Run`:

```go
func (q *QueueAggregator) Run(ctx context.Context, chs []<-chan Session) <-chan ProjectSessions {
	out := make(chan ProjectSessions)
	sessions := make(map[string][]Session)
	mu := sync.Mutex{}

	// Объединённый входной канал
	in := make(chan Session)

	// Сливаем все входные каналы в один
	for _, ch := range chs {
		ch := ch
		go func() {
			for session := range ch {
				select {
				case in <- session:
				case <-ctx.Done():
					return
				}
			}
		}()
	}

	// Периодическая очистка по таймеру
	go func() {
		ticker := time.NewTicker(q.cfg.FlushInterval)
		defer ticker.Stop()
		defer close(out)

		for {
			select {
			case session := <-in:
				mu.Lock()
				buf := sessions[session.ProjectID]
				buf = append(buf, session)
				sessions[session.ProjectID] = buf

				if len(buf) >= q.cfg.MaxSessionsPerProject {
					out <- ProjectSessions{
						ProjectID: session.ProjectID,
						Sessions:  buf,
					}
					delete(sessions, session.ProjectID)
				}
				mu.Unlock()

			case <-ticker.C:
				mu.Lock()
				for projectID, sess := range sessions {
					if len(sess) > 0 {
						out <- ProjectSessions{
							ProjectID: projectID,
							Sessions:  sess,
						}
					}
				}
				sessions = make(map[string][]Session)
				mu.Unlock()

			case <-ctx.Done():
				// При отмене: отправим всё, что осталось
				mu.Lock()
				for projectID, sess := range sessions {
					if len(sess) > 0 {
						out <- ProjectSessions{
							ProjectID: projectID,
							Sessions:  sess,
						}
					}
				}
				mu.Unlock()
				return
			}
		}
	}()

	return out
}
```

---

### 🧠 Как это работает:

- **`in`** собирает все сессии в один канал из входных
    
- Внутри горутины:
    
    - по каждой новой сессии она добавляется в `map[ProjectID][]Session`
        
    - если длина достигла лимита → отправляется в `out`, и очищается
        
    - по таймеру отправляются **все собранные**, даже если лимит не достигнут
        
    - при завершении `ctx` всё оставшееся тоже уходит в `out`

### 📌 Пример использования

```go
ctx, cancel := context.WithCancel(context.Background())
defer cancel()

chs := []<-chan Session{...}
agg := NewQueueAggregator(QueueAggregatorCfg{
	FlushInterval:         5 * time.Second,
	MaxSessionsPerProject: 100,
})

for ps := range agg.Run(ctx, chs) {
	fmt.Println("Flushed project:", ps.ProjectID, "count:", len(ps.Sessions))
}
```

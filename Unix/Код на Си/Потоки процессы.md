
Вот развернутый пример работы с процессами и потоками в Linux на языке C с подробными комментариями:

---

## **1. Работа с процессами (fork, exec, wait)**

### 🔹 **Создание процесса (fork)**
```c
#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>

int main() {
    pid_t pid = fork(); // Создаем новый процесс

    if (pid == -1) {
        perror("fork failed");
        return 1;
    }

    if (pid == 0) {
        // Код выполняется в дочернем процессе
        printf("Child process: PID = %d, Parent PID = %d\n", 
               getpid(), getppid());
        sleep(2); // Имитация работы
        printf("Child process exiting\n");
        return 42; // Выход с кодом 42
    } else {
        // Код выполняется в родительском процессе
        printf("Parent process: PID = %d, Child PID = %d\n", 
               getpid(), pid);
        
        int status;
        waitpid(pid, &status, 0); // Ожидание завершения дочернего процесса
        
        if (WIFEXITED(status)) {
            printf("Child exited with status: %d\n", WEXITSTATUS(status));
        }
    }

    return 0;
}
```

**Пояснения:**
- `fork()` создает копию процесса (дочерний процесс)
- В дочернем процессе `fork()` возвращает `0`
- В родительском процессе `fork()` возвращает PID дочернего
- `waitpid()` приостанавливает родительский процесс до завершения дочернего

## **2. Запуск новой программы (exec)**
```c
#include <unistd.h>
#include <stdio.h>

int main() {
    char *args[] = {"/bin/ls", "-l", NULL}; // Аргументы для ls
    
    printf("Before exec\n");
    
    // Заменяем текущий процесс на ls
    execvp(args[0], args);
    
    // Сюда попадем только при ошибке
    perror("exec failed");
    return 1;
}
```

**Варианты функций exec:**
- `execl()` - принимает список аргументов
- `execv()` - принимает массив аргументов
- `execvp()` - ищет программу в PATH

## **3. Работа с потоками (pthread)**
```c
#include <pthread.h>
#include <stdio.h>
#include <unistd.h>

// Функция, выполняемая в потоке
void* thread_func(void* arg) {
    int thread_num = *(int*)arg;
    printf("Thread %d started\n", thread_num);
    sleep(1);
    printf("Thread %d finished\n", thread_num);
    return NULL;
}

int main() {
    pthread_t tid1, tid2;
    int num1 = 1, num2 = 2;

    // Создаем два потока
    pthread_create(&tid1, NULL, thread_func, &num1);
    pthread_create(&tid2, NULL, thread_func, &num2);

    // Ожидаем завершения потоков
    pthread_join(tid1, NULL);
    pthread_join(tid2, NULL);

    printf("All threads completed\n");
    return 0;
}
```

**Ключевые функции pthread:**
- `pthread_create()` - создает новый поток
- `pthread_join()` - ожидает завершения потока
- `pthread_exit()` - завершает текущий поток

## **4. Синхронизация потоков (мьютексы)**
```c
#include <pthread.h>
#include <stdio.h>

int counter = 0;
pthread_mutex_t mutex = PTHREAD_MUTEX_INITIALIZER;

void* increment(void* arg) {
    for (int i = 0; i < 100000; i++) {
        pthread_mutex_lock(&mutex);
        counter++;
        pthread_mutex_unlock(&mutex);
    }
    return NULL;
}

int main() {
    pthread_t t1, t2;
    
    pthread_create(&t1, NULL, increment, NULL);
    pthread_create(&t2, NULL, increment, NULL);
    
    pthread_join(t1, NULL);
    pthread_join(t2, NULL);
    
    printf("Final counter value: %d\n", counter);
    return 0;
}
```

**Пояснение:**
- Мьютекс защищает общий ресурс (counter) от одновременного доступа
- Без мьютекса возникла бы гонка данных (race condition)

## **5. Совместное использование fork и pthread**
```c
#include <pthread.h>
#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>

void* thread_func(void* arg) {
    printf("Thread in process %d\n", getpid());
    return NULL;
}

int main() {
    pthread_t tid;
    pthread_create(&tid, NULL, thread_func, NULL);
    
    pid_t pid = fork();
    
    if (pid == 0) {
        printf("Child process %d\n", getpid());
        pthread_create(&tid, NULL, thread_func, NULL);
    } else {
        wait(NULL);
    }
    
    pthread_join(tid, NULL);
    return 0;
}
```

**Важно:** После fork() в дочернем процессе будет копия всех потоков родителя, но выполняется только один поток.

## **Как компилировать**
Для программ с потоками нужно добавить `-pthread`:
```bash
gcc process.c -o process
gcc thread.c -o thread -pthread
```

## **Ключевые отличия процессов и потоков**
| Характеристика      | Процессы                   | Потоки                    |
|---------------------|---------------------------|--------------------------|
| Память              | Изолированная             | Общая                    |
| Создание            | `fork()`                  | `pthread_create()`       |
| Коммуникация        | Каналы, IPC               | Глобальные переменные    |
| Переключение контекста | Дорогое                | Дешевое                  |
| Отказоустойчивость  | Один процесс может упасть | Падение потока убивает все |

Выбор между процессами и потоками зависит от конкретной задачи. Потоки эффективнее для CPU-bound задач, процессы - для изоляции.

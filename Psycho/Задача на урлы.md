Вот асинхронный метод на Go, который делает HTTP-запросы к списку URL и выводит результат (OK или НЕ ОК) в зависимости от статуса:

```go
package main

import (
	"fmt"
	"net/http"
	"sync"
)

// fetchURL делает GET-запрос к URL и возвращает статус и ошибку
func fetchURL(url string) (string, error) {
	resp, err := http.Get(url)
	if err != nil {
		return "", err
	}
	defer resp.Body.Close()

	if resp.StatusCode == http.StatusOK {
		return "OK", nil
	}
	return fmt.Sprintf("НЕ ОК (Status: %d)", resp.StatusCode), nil
}

// processURLs асинхронно обрабатывает слайс URL и выводит результаты
func processURLs(urls []string) {
	var wg sync.WaitGroup
	results := make(chan string, len(urls))

	for _, url := range urls {
		wg.Add(1)
		go func(u string) {
			defer wg.Done()
			status, err := fetchURL(u)
			if err != nil {
				results <- fmt.Sprintf("НЕ ОК: %s | Error: %v", u, err)
			} else {
				results <- fmt.Sprintf("%s: %s | URL: %s", status, "всё хорошо", u)
			}
		}(url)
	}

	// Закрываем канал после завершения всех горутин
	go func() {
		wg.Wait()
		close(results)
	}()

	// Выводим результаты
	for res := range results {
		fmt.Println(res)
	}
}

func main() {
	urls := []string{
		"https://httpbin.org/status/200",
		"https://httpbin.org/status/404",
		"https://httpbin.org/status/500",
		"https://nonexistent.example.com", // Несуществующий URL
	}

	processURLs(urls)
}
```

### **Как это работает?**
1. **`fetchURL(url)`**:
   - Делает GET-запрос.
   - Если статус `200` → возвращает `"OK"`.
   - Если статус не `200` → возвращает `"НЕ ОК (Status: ...)"`.
   - Если ошибка (например, нет соединения) → возвращает ошибку.

2. **`processURLs(urls)`**:
   - Запускает горутины для каждого URL.
   - Использует `sync.WaitGroup` для ожидания завершения всех запросов.
   - Результаты пишутся в канал `results` и выводятся построчно.

3. **Вывод в консоль**:
   - Если статус `200` → `OK: всё хорошо | URL: ...`
   - Если статус не `200` → `НЕ ОК (Status: 404): ... | URL: ...`
   - Если ошибка → `НЕ ОК: https://... | Error: ...`

### **Пример вывода**
```
OK: всё хорошо | URL: https://httpbin.org/status/200
НЕ ОК (Status: 404) | URL: https://httpbin.org/status/404
НЕ ОК (Status: 500) | URL: https://httpbin.org/status/500
НЕ ОК: https://nonexistent.example.com | Error: Get "https://nonexistent.example.com": dial tcp: lookup nonexistent.example.com: no such host
```

### **Оптимизации**
- Можно добавить таймаут (`http.Client{ Timeout: 5 * time.Second }`).
- Можно ограничить количество одновременных запросов (семафором или пулом горутин).  
- Можно парсить тело ответа, если нужно (`ioutil.ReadAll(resp.Body)`).  

С воркер пуллом:

Я добавил в ваш код паттерн Worker Pool, который ограничивает количество одновременно работающих горутин. Это поможет избежать проблем с ресурсами при обработке большого числа URL-адресов.

Вот модифицированный код:

```go
package main

import (
	"fmt"
	"net/http"
	"sync"
)

// fetchURL делает GET-запрос к URL и возвращает статус и ошибку
func fetchURL(url string) (string, error) {
	resp, err := http.Get(url)
	if err != nil {
		return "", err
	}
	defer resp.Body.Close()

	if resp.StatusCode == http.StatusOK {
		return "OK", nil
	}
	return fmt.Sprintf("НЕ ОК (Status: %d)", resp.StatusCode), nil
}

// worker получает URL из канала jobs и отправляет результаты в канал results
func worker(id int, wg *sync.WaitGroup, jobs <-chan string, results chan<- string) {
	defer wg.Done()
	for url := range jobs {
		status, err := fetchURL(url)
		if err != nil {
			results <- fmt.Sprintf("Worker %d: НЕ ОК: %s | Error: %v", id, url, err)
		} else {
			results <- fmt.Sprintf("Worker %d: %s: %s | URL: %s", id, status, "всё хорошо", url)
		}
	}
}

// processURLsWithWorkerPool обрабатывает слайс URL с использованием пула воркеров
func processURLsWithWorkerPool(urls []string, numWorkers int) {
	var wg sync.WaitGroup
	jobs := make(chan string, len(urls))
	results := make(chan string, len(urls))

	// Запускаем воркеры
	for w := 1; w <= numWorkers; w++ {
		wg.Add(1)
		go worker(w, &wg, jobs, results)
	}

	// Отправляем URL в канал jobs
	for _, url := range urls {
		jobs <- url
	}
	close(jobs)

	// Закрываем канал results после завершения всех воркеров
	go func() {
		wg.Wait()
		close(results)
	}()

	// Выводим результаты
	for res := range results {
		fmt.Println(res)
	}
}

func main() {
	urls := []string{
		"https://httpbin.org/status/200",
		"https://httpbin.org/status/404", 
		"https://httpbin.org/status/500",
		"https://httpbin.org/status/201",
		"https://httpbin.org/status/202",
		"https://httpbin.org/status/400",
		"https://nonexistent.example.com", // Несуществующий URL
	}

	// Обрабатываем URL с использованием пула из 3 воркеров
	processURLsWithWorkerPool(urls, 3)
}
```

### 🔧 Ключевые изменения в коде

1.  **Функция `worker`**: Каждый воркер постоянно читает URL из канала `jobs`, обрабатывает их с помощью `fetchURL` и отправляет результат в канал `results`. Воркеры завершают работу автоматически, когда канал `jobs` закрыт и пуст.
2.  **Управление горутинами**: Количество одновременно работающих горутин теперь ограничено значением `numWorkers` (в данном случае 3). Это предотвращает создание неограниченного количества горутин и потенциальное исчерпание ресурсов.
3.  **Синхронизация**: `sync.WaitGroup` используется для ожидания завершения работы всех воркеров перед закрытием канала `results`.

### 💡 Преимущества Worker Pool

-   **Контроль нагрузки**: Вы можете точно контролировать, сколько запросов выполняется одновременно.
-   **Эффективность ресурсов**: Избегает создания чрезмерного количества горутин, особенно при обработке тысяч URL.
-   **Улучшенная обработка ошибок**: Структура пула облегчает добавление механизмов повторных попыток (retry logic) и обработки ошибок.

Если вам нужно обрабатывать десятки тысяч URL, рассмотрите возможность использования готовых библиотек, таких как `pond`, которые предлагают дополнительные функции, такие как автоматическое масштабирование и обработку паник.

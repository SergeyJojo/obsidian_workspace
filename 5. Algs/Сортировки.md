
### **Quick Sort (Быстрая сортировка) на Go**  
Использует стратегию "разделяй и властвуй" с опорным элементом (pivot).

```go
package main

import "fmt"

func quickSort(arr []int) {
    if len(arr) < 2 {
        return
    }
    
	left, right := 0, len(arr)-1

    pivot := arr[len(arr)/2]
    
    for left <= right {
        for arr[left] < pivot {
            left++
        }

        for arr[right] > pivot {
            right--
        }

        if left <= right {
            arr[left], arr[right] = arr[right], arr[left]

            left++
            right--
        }
    }

  

    if right > 0 {
        quickSort(arr[:right+1])
    }
    
    if left < len(arr)-1 {
        quickSort(arr[left:])
    }
}

func main() {
    arr := []int{3, 6, 8, 10, 1, 2, 1}
    sorted := quickSort(arr)
    fmt.Println(sorted)  // [1 1 2 3 6 8 10]
}
```

### **Shell Sort (Сортировка Шелла) на Go**  
Оптимизированная версия сортировки вставками с уменьшающимся интервалом.

```go
package main

import "fmt"

func shellSort(arr []int) {
    n := len(arr)
    for gap := n / 2; gap > 0; gap /= 2 {
        for i := gap; i < n; i++ {
            temp := arr[i]
            j := i
            for j >= gap && arr[j-gap] > temp {
                arr[j] = arr[j-gap]
                j -= gap
            }
            arr[j] = temp
        }
    }
}

func main() {
    arr := []int{3, 6, 8, 10, 1, 2, 1}
    shellSort(arr)
    fmt.Println(arr)  // [1 1 2 3 6 8 10]
}
```

### **Ключевые различия**
| Алгоритм       | Сложность       | Память   | Особенности                     |
|----------------|----------------|----------|----------------------------------|
| **Quick Sort** | O(n log n)     | O(log n) | Быстрый, но нестабильный        |
| **Shell Sort** | O(n log² n)    | O(1)     | Медленнее, но работает на месте |

**Quick Sort** обычно быстрее, но **Shell Sort** не требует дополнительной памяти. Оба алгоритма используют слайсы (slices) в Go, которые передаются по ссылке.



### 1. **Пузырьковая сортировка (Bubble Sort)**

```go
package main

import "fmt"

func bubbleSort(arr []int) {
	n := len(arr)
	for i := 0; i < n; i++ {
		for j := 0; j < n-i-1; j++ {
			if arr[j] > arr[j+1] {
				arr[j], arr[j+1] = arr[j+1], arr[j] // swap
			}
		}
	}
}

func main() {
	arr := []int{5, 2, 9, 1, 5, 6}
	bubbleSort(arr)
	fmt.Println(arr) // [1 2 5 5 6 9]
}
```

### 2. **Сортировка вставками (Insertion Sort)**

```go
package main

import "fmt"

func insertionSort(arr []int) {
	for i := 1; i < len(arr); i++ {
		key := arr[i]
		j := i - 1
		for j >= 0 && arr[j] > key {
			arr[j+1] = arr[j]
			j--
		}
		arr[j+1] = key
	}
}

func main() {
	arr := []int{5, 2, 9, 1, 5, 6}
	insertionSort(arr)
	fmt.Println(arr) // [1 2 5 5 6 9]
}
```

### 3. **Сортировка выбором (Selection Sort)**

```go
package main

import "fmt"

func selectionSort(arr []int) {
	n := len(arr)
	for i := 0; i < n-1; i++ {
		minIndex := i
		for j := i + 1; j < n; j++ {
			if arr[j] < arr[minIndex] {
				minIndex = j
			}
		}
		arr[i], arr[minIndex] = arr[minIndex], arr[i]
	}
}

func main() {
	arr := []int{5, 2, 9, 1, 5, 6}
	selectionSort(arr)
	fmt.Println(arr) // [1 2 5 5 6 9]
}
```

### 4. **Быстрая сортировка (Quick Sort)**

```go
package main

import "fmt"

func quickSort(arr []int) []int {
	if len(arr) < 2 {
		return arr
	}

	pivot := arr[0]
	left := []int{}
	right := []int{}

	for _, v := range arr[1:] {
		if v < pivot {
			left = append(left, v)
		} else {
			right = append(right, v)
		}
	}

	return append(append(quickSort(left), pivot), quickSort(right)...)
}

func main() {
	arr := []int{5, 2, 9, 1, 5, 6}
	sorted := quickSort(arr)
	fmt.Println(sorted) // [1 2 5 5 6 9]
}
```

### 5. **Сортировка слиянием (Merge Sort)**

```go
package main

import "fmt"

func mergeSort(arr []int) []int {
	if len(arr) < 2 {
		return arr
	}
	mid := len(arr) / 2
	left := mergeSort(arr[:mid])
	right := mergeSort(arr[mid:])
	return merge(left, right)
}

func merge(left, right []int) []int {
	result := []int{}
	i, j := 0, 0
	for i < len(left) && j < len(right) {
		if left[i] < right[j] {
			result = append(result, left[i])
			i++
		} else {
			result = append(result, right[j])
			j++
		}
	}
	result = append(result, left[i:]...)
	result = append(result, right[j:]...)
	return result
}

func main() {
	arr := []int{5, 2, 9, 1, 5, 6}
	sorted := mergeSort(arr)
	fmt.Println(sorted) // [1 2 5 5 6 9]
}
```

### 6. **Сортировка кучей (Heap Sort)**

```go
package main

import "fmt"

func heapify(arr []int, n, i int) {
	largest := i
	left := 2*i + 1
	right := 2*i + 2

	if left < n && arr[left] > arr[largest] {
		largest = left
	}
	if right < n && arr[right] > arr[largest] {
		largest = right
	}
	if largest != i {
		arr[i], arr[largest] = arr[largest], arr[i]
		heapify(arr, n, largest)
	}
}

func heapSort(arr []int) {
	n := len(arr)
	for i := n / 2 - 1; i >= 0; i-- {
		heapify(arr, n, i)
	}
	for i := n - 1; i >= 0; i-- {
		arr[0], arr[i] = arr[i], arr[0]
		heapify(arr, i, 0)
	}
}

func main() {
	arr := []int{5, 2, 9, 1, 5, 6}
	heapSort(arr)
	fmt.Println(arr) // [1 2 5 5 6 9]
}
```

### 7. **Сортировка подсчётом (Counting Sort)**

```go
package main

import "fmt"

func countingSort(arr []int) []int {
	max := arr[0]
	for _, num := range arr {
		if num > max {
			max = num
		}
	}

	count := make([]int, max+1)
	for _, num := range arr {
		count[num]++
	}

	sorted := []int{}
	for i, c := range count {
		for c > 0 {
			sorted = append(sorted, i)
			c--
		}
	}
	return sorted
}

func main() {
	arr := []int{5, 2, 9, 1, 5, 6}
	sorted := countingSort(arr)
	fmt.Println(sorted) // [1 2 5 5 6 9]
}
```

### 8. **Поразрядная сортировка (Radix Sort)**

```go
package main

import "fmt"

func getMax(arr []int) int {
	max := arr[0]
	for _, num := range arr {
		if num > max {
			max = num
		}
	}
	return max
}

func countingSortForRadix(arr []int, exp int) []int {
	count := make([]int, 10)
	output := make([]int, len(arr))

	for _, num := range arr {
		index := (num / exp) % 10
		count[index]++
	}

	for i := 1; i < 10; i++ {
		count[i] += count[i-1]
	}

	for i := len(arr) - 1; i >= 0; i-- {
		num := arr[i]
		index := (num / exp) % 10
		output[count[index]-1] = num
		count[index]--
	}

	return output
}

func radixSort(arr []int) []int {
	max := getMax(arr)
	for exp := 1; max/exp > 0; exp *= 10 {
		arr = countingSortForRadix(arr, exp)
	}
	return arr
}

func main() {
	arr := []int{170, 45, 75, 90, 802, 24, 2, 66}
	sorted := radixSort(arr)
	fmt.Println(sorted) // [2 24 45 66 75 90 170 802]
}
```

### 9. **Тимсорт (Timsort)**

Go не имеет встроенной реализации Тимсорта, но его можно использовать с библиотеками (например, в Python, Java). Однако Go использует **Quick Sort** в стандартной библиотеке для сортировки.

### 10. **Сортировка Шелла (Shell Sort)**

```go
package main

import "fmt"

func shellSort(arr []int) {
	n := len(arr)
	gap := n / 2
	for gap > 0 {
		for i := gap; i < n; i++ {
			temp := arr[i]
			j := i
			for j >= gap && arr[j-gap] > temp {
				arr[j] = arr[j-gap]
				j -= gap
			}
			arr[j] = temp
		}
		gap /= 2
	}
}

func main() {
	arr := []int{5, 2, 9, 1, 5, 6}
	shellSort(arr)
	fmt.Println(arr) // [1 2 5 5 6 9]
}
```

Это различные алгоритмы сортировки, каждый из которых имеет свои особенности и применяется в зависимости от требований и характеристик данных. Для большинства практических задач **быстрая сортировка**, **сортировка слиянием** и **сортировка кучей** являются наиболее эффективными.

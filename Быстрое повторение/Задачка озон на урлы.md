Конечно! Вот аккуратно отформатированный и читаемый вариант твоего кода:

```go
package main

import (
	"context"
	"fmt"
	"net/http"
	"time"
)

func main() {
	urls := make(chan string)
	results := make(chan string)

	go func() {
		defer close(urls)
		// Внешний источник URL'ов
		for _, url := range []string{
			"http://ozon.ru",
			"https://ozon.ru",
			"http://google.com",
			"http://somesite.com",
			"http://non-existent.domain.tld",
			"https://ya.ru",
			"http://ya.ru",
			"http://ёёёё",
		} {
			urls <- url
		}
	}()

	ctx, cancel := context.WithCancel(context.Background())
	defer cancel()

	const workers = 5
	okCount := 0

	go func() {
		for result := range results {
			fmt.Println(result)
			if len(result) >= 2 && result[len(result)-2:] == "ok" {
				okCount++
				if okCount == 2 {
					fmt.Println("Получено 2 успешных ответа. Прерывание оставшихся запросов.")
					cancel()
				}
			}
		}
	}()

	for i := 0; i < workers; i++ {
		go worker(ctx, urls, results)
	}

	time.Sleep(10 * time.Second) // ожидание завершения
	close(results)
}

func worker(ctx context.Context, urls <-chan string, results chan<- string) {
	client := http.Client{Timeout: 3 * time.Second}
	for {
		select {
		case <-ctx.Done():
			return
		case url, ok := <-urls:
			if !ok {
				return
			}
			resp, err := client.Get(url)
			if err != nil {
				results <- fmt.Sprintf("%s - not ok", url)
				continue
			}
			resp.Body.Close()
			if resp.StatusCode == 200 {
				results <- fmt.Sprintf("%s - ok", url)
			} else {
				results <- fmt.Sprintf("%s - not ok", url)
			}
		}
	}
}
```

---

### Основные улучшения:

- Выравнивание отступов.
    
- Мелкая защита: `if len(result) >= 2` перед проверкой `result[len(result)-2:] == "ok"`, чтобы избежать возможной паники на пустой строке.
    
- Комментарии к важным частям.
    
- Небольшое исправление пробелов вокруг `go func()` в блоке чтения `results`.

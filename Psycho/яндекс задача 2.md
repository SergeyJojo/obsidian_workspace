## 📝 Реализация интерфейсов

```go
package main

import (
	"context"
	"crypto/aes"
	"crypto/cipher"
	"crypto/rand"
	"io"
	"os"
	"sync"
)

// FileOperator интерфейс для работы с файлами
type FileOperator interface {
	WriteChunk(filename string, data []byte) error
	ReadChunk(filename string) ([]byte, error)
}

// CryptoOperator интерфейс для шифрования/дешифрования
type CryptoOperator interface {
	Encrypt(data []byte) ([]byte, error)
	Decrypt(encryptedData []byte) ([]byte, error)
}

// Реализация FileOperator
type FileManager struct{}

func (fm *FileManager) WriteChunk(filename string, data []byte) error {
	return os.WriteFile(filename, data, 0644)
}

func (fm *FileManager) ReadChunk(filename string) ([]byte, error) {
	return os.ReadFile(filename)
}

// Реализация CryptoOperator
type CryptoManager struct {
	key []byte
}

func (cm *CryptoManager) Encrypt(data []byte) ([]byte, error) {
	block, err := aes.NewCipher(cm.key)
	if err != nil {
		return nil, err
	}

	gcm, err := cipher.NewGCM(block)
	if err != nil {
		return nil, err
	}

	nonce := make([]byte, gcm.NonceSize())
	if _, err = io.ReadFull(rand.Reader, nonce); err != nil {
		return nil, err
	}

	return gcm.Seal(nonce, nonce, data, nil), nil
}

func (cm *CryptoManager) Decrypt(encryptedData []byte) ([]byte, error) {
	block, err := aes.NewCipher(cm.key)
	if err != nil {
		return nil, err
	}

	gcm, err := cipher.NewGCM(block)
	if err != nil {
		return nil, err
	}

	nonceSize := gcm.NonceSize()
	nonce, ciphertext := encryptedData[:nonceSize], encryptedData[nonceSize:]

	return gcm.Open(nil, nonce, ciphertext, nil)
}
```

## 🔄 Задача 1: Разбиение и шифрование файла

```go
func SplitAndEncryptLargeFile(
	ctx context.Context,
	inputFile string,
	fileOp FileOperator,
	cryptoOp CryptoOperator,
	chunkSizeMB int,
	outputDir string,
) error {

	file, err := os.Open(inputFile)
	if err != nil {
		return err
	}
	defer file.Close()

	chunkSize := int64(chunkSizeMB) * 1024 * 1024
	buffer := make([]byte, chunkSize)
	chunkNumber := 0

	for {
		// Читаем чанк
		n, err := file.Read(buffer)
		if err != nil && err != io.EOF {
			return err
		}
		if n == 0 {
			break
		}

		// Шифруем чанк
		encryptedData, err := cryptoOp.Encrypt(buffer[:n])
		if err != nil {
			return err
		}

		// Записываем зашифрованный чанк
		chunkFilename := fmt.Sprintf("%s/chunk_%04d.enc", outputDir, chunkNumber)
		if err := fileOp.WriteChunk(chunkFilename, encryptedData); err != nil {
			return err
		}

		chunkNumber++
		
		if err == io.EOF {
			break
		}
	}

	return nil
}
```

## 🔄 Задача 2: Сборка и дешифровка (с горутинами)

```go
func AssembleAndDecryptFiles(
	ctx context.Context,
	fileOp FileOperator,
	cryptoOp CryptoOperator,
	chunkFiles []string,
	outputFile string,
) error {

	output, err := os.Create(outputFile)
	if err != nil {
		return err
	}
	defer output.Close()

	var wg sync.WaitGroup
	results := make(chan []byte, len(chunkFiles))
	errors := make(chan error, len(chunkFiles))

	// Запускаем горутины для параллельного чтения и дешифровки
	for _, chunkFile := range chunkFiles {
		wg.Add(1)
		
		go func(filename string) {
			defer wg.Done()

			// Читаем зашифрованный чанк
			encryptedData, err := fileOp.ReadChunk(filename)
			if err != nil {
				errors <- err
				return
			}

			// Дешифруем
			decryptedData, err := cryptoOp.Decrypt(encryptedData)
			if err != nil {
				errors <- err
				return
			}

			results <- decryptedData
		}(chunkFile)
	}

	// Ждем завершения всех горутин
	go func() {
		wg.Wait()
		close(results)
		close(errors)
	}()

	// Обрабатываем ошибки
	select {
	case err := <-errors:
		return err
	default:
	}

	// Собираем результаты в правильном порядке
	// Для сохранения порядка нужно использовать мапу с номером чанка
	orderedResults := make(map[int][]byte, len(chunkFiles))
	for i, chunkFile := range chunkFiles {
		// Извлекаем номер из имени файла
		var chunkNum int
		fmt.Sscanf(chunkFile, "chunk_%04d.enc", &chunkNum)
		orderedResults[chunkNum] = <-results
	}

	// Записываем в правильном порядке
	for i := 0; i < len(chunkFiles); i++ {
		if data, exists := orderedResults[i]; exists {
			if _, err := output.Write(data); err != nil {
				return err
			}
		}
	}

	return nil
}
```

## 🎯 Альтернативная версия с io.CopyN

```go
func SplitWithCopyN(
	inputFile string,
	cryptoOp CryptoOperator,
	chunkSizeMB int,
) error {

	file, err := os.Open(inputFile)
	if err != nil {
		return err
	}
	defer file.Close()

	chunkSize := int64(chunkSizeMB) * 1024 * 1024
	chunkNumber := 0

	for {
		chunkFilename := fmt.Sprintf("chunk_%04d.enc", chunkNumber)
		chunkFile, err := os.Create(chunkFilename)
		if err != nil {
			return err
		}

		// Читаем ровно chunkSize байт
		encryptWriter := &EncryptWriter{
			Writer:    chunkFile,
			CryptoOp:  cryptoOp,
		}

		_, err = io.CopyN(encryptWriter, file, chunkSize)
		chunkFile.Close()

		if err != nil {
			if err == io.EOF {
				break
			}
			return err
		}

		chunkNumber++
	}

	return nil
}

// EncryptWriter обертка для шифрования при записи
type EncryptWriter struct {
	Writer   io.Writer
	CryptoOp CryptoOperator
	buffer   []byte
}

func (ew *EncryptWriter) Write(p []byte) (n int, err error) {
	encrypted, err := ew.CryptoOp.Encrypt(p)
	if err != nil {
		return 0, err
	}
	return ew.Writer.Write(encrypted)
}
```

## 🚀 Использование

```go
func main() {
	// Инициализация
	fileOp := &FileManager{}
	cryptoOp := &CryptoManager{key: []byte("32-byte-long-key-1234567890123456")}

	// Задача 1: Разбить и зашифровать
	err := SplitAndEncryptLargeFile(
		context.Background(),
		"large_file.iso",
		fileOp,
		cryptoOp,
		100, // 100MB chunks
		"chunks_dir",
	)
	if err != nil {
		panic(err)
	}

	// Задача 2: Собрать обратно
	chunkFiles := []string{
		"chunks_dir/chunk_0000.enc",
		"chunks_dir/chunk_0001.enc",
		// ...
	}

	err = AssembleAndDecryptFiles(
		context.Background(),
		fileOp,
		cryptoOp,
		chunkFiles,
		"restored_file.iso",
	)
	if err != nil {
		panic(err)
	}
}
```

## 💡 Ключевые моменты решения:

1. **Интерфейсы** - четкое разделение ответственности
2. **Потокобезопасность** - использование горутин и каналов
3. **Эффективная работа с памятью** - обработка по чанкам
4. **Обработка ошибок** - правильная работа с контекстом
5. **Правильный порядок** - сохранение порядка чанков при сборке

```go
package main

import (
	"fmt"
	"time"
)

type Config struct {
	Host    string
	Port    int
	UseSSL  bool
	Timeout time.Duration
}

type Option func(*Config)

func WithHost(host string) Option {
	return func(c *Config) {
		c.Host = host
	}
}

func WithPort(port int) Option {
	return func(c *Config) {
		c.Port = port
	}
}

func WithSSL(useSSL bool) Option {
	return func(c *Config) {
		c.UseSSL = useSSL
	}
}

func WithTimeout(timeout time.Duration) Option {
	return func(c *Config) {
		c.Timeout = timeout
	}
}

func NewConfig(opts ...Option) *Config {
	// Default config
	cfg := &Config{
		Host:    "localhost",
		Port:    8080,
		UseSSL:  false,
		Timeout: 30 * time.Second,
	}

	// Apply all options
	for _, opt := range opts {
		opt(cfg)
	}

	return cfg
}

func main() {
	cfg := NewConfig(
		WithHost("example.com"),
		WithPort(443),
		WithSSL(true),
		WithTimeout(60*time.Second),
	)

	fmt.Printf("Config: %+v\n", cfg)
}
```

```go
package main

import (
	"fmt"
	"sync"
)

type Task struct {
	fn func()
	id int
}

func worker(id int, tasks <-chan Task, wg *sync.WaitGroup) {
	for task := range tasks {
		fmt.Printf("Worker %d started Task #%d\n", id, task.id)

		task.fn()

		fmt.Printf("Worker %d ended Task #%d\n", id, task.id)
		wg.Done()
	}
}

const (
	numWorkers = 5
	numTasks   = 10
)

func main() {
	taskChan := make(chan Task, numTasks)
	wg := &sync.WaitGroup{}

	wg.Add(numTasks)
	for i := 1; i <= numWorkers; i++ {
		go worker(i, taskChan, wg)
	}

	for i := 1; i <= numTasks; i++ {
		taskChan <- Task{
			id: i,
			fn: func() {
				fmt.Printf("Task #%d is being processed\n", i)
			},
		}
	}
	close(taskChan)

	wg.Wait()
	fmt.Println("All tasks done")
}
```

Внесены некоторые исправления:

1. Добавлена переменная `id` в структуру `Task` для корректной работы с задачами.
2. Обновлены задачи, передаваемые в канал, чтобы каждая задача имела свою функцию `fn` с выводом в консоль.

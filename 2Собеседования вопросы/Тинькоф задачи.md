## 🧠 Пример 1: Работа с указателями и `append`

```go
package main

import (
	"fmt"
)

func appendLenWrong(numbers []*int) {
	size := len(numbers) // 3
	numbers = append(numbers, &size) // локальное изменение — не влияет на внешний слайс
}

func appendLenGoodFirst(numbers *[]*int) {
	size := len(*numbers)
	*numbers = append(*numbers, &size) // мутирует оригинальный слайс
}

func appendLenGoodSecond(numbers []*int) []*int {
	size := len(numbers)
	numbers = append(numbers, &size)
	return numbers
}

func main() {
	numbers := make([]*int, 0, 5)
	var number int

	for range 3 {
		number++
		numbers = append(numbers, &number)
	}

	// Все указатели ссылаются на одну и ту же переменную — в итоге три раза 3
	for _, number := range numbers {
		fmt.Printf("%d ", *number)
	}
}
```

### Вывод:

```
3 3 3
```

### Проблема:

- В цикле `range 3` мы каждый раз добавляем адрес _одной и той же_ переменной `number`, просто с изменённым значением.
    
- Поэтому все элементы указывают на одно место в памяти.

## 🔄 Пример 2: Ошибка мутации общей мапы в горутинах

```go
package main

import (
	"fmt"
	"sync"
	"time"
)

func RunWriter() <-chan map[string]int {
	var prices = make(chan map[string]int)

	go func() {
		currentPrice := map[string]int{
			"AAPL": 163,
			"USD":  117,
			"EUR":  124,
			"NVDA": 234,
		}

		for i := 1; i < 5; i++ {
			for key, value := range currentPrice {
				currentPrice[key] = int(float64(value) * 1.3)
			}
			// ❌ Ошибка: передаётся *та же самая мапа* каждый раз!
			// Надо делать копию
			copyMap := make(map[string]int)
			for k, v := range currentPrice {
				copyMap[k] = v
			}
			prices <- copyMap
			time.Sleep(100 * time.Millisecond)
		}
		close(prices)
	}()
	return prices
}

func RunProcessor(wg *sync.WaitGroup, prices []map[string]int, mu *sync.Mutex) {
	go func() {
		defer wg.Done()
		for _, price := range prices {
			mu.Lock()
			for key, value := range price {
				price[key] = value + 1
			}
			fmt.Println(price)
			mu.Unlock()
		}
	}()
}

func main() {
	p := RunWriter()

	var prices []map[string]int
	for price := range p {
		prices = append(prices, price)
	}

	fmt.Println("Original prices:")
	for _, price := range prices {
		fmt.Println(price)
	}

	wg := &sync.WaitGroup{}
	mu := &sync.Mutex{}

	for i := 0; i < 10; i++ {
		wg.Add(1)
		RunProcessor(wg, prices, mu)
	}

	wg.Wait()
}
```

### Проблема:

- Без копирования мапы в `RunWriter()` в канал попадает _одна и та же_ мапа.

## ⚙️ Пример 3: Горутинки и каналы

```go
package main

import (
	"fmt"
	"strconv"
	"sync"
)

func main() {
	var wg sync.WaitGroup
	ch := make(chan string, 3)

	for i := 0; i < 5; i++ {
		wg.Add(1)
		go func(i int) {
			defer wg.Done()
			ch <- fmt.Sprintf("Goroutine %s", strconv.Itoa(i))
		}(i)
	}

	// Ждём завершения всех горутин и закрываем канал
	go func() {
		wg.Wait()
		close(ch)
	}()

	// Чтение из канала
	for msg := range ch {
		fmt.Println(msg)
	}
}
```

### Ключевые моменты:

- `sync.WaitGroup` и закрытие канала в горутине, чтобы избежать deadlock.
    
- Не нужно дополнительной синхронизации (`Mutex`) при записи в канал — каналы в Go потокобезопасны.

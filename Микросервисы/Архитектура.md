```go
package kafka

import (
	"context"
	"github.com/segmentio/kafka-go"
	"log"
)

func PublishMessage(topic, key, message string) error {
	writer := kafka.Writer{
		Addr:     kafka.TCP("localhost:9092"),
		Topic:    topic,
		Balancer: &kafka.LeastBytes{},
	}

	defer writer.Close()

	err := writer.WriteMessages(context.Background(), kafka.Message{
		Key:   []byte(key),
		Value: []byte(message),
	})

	if err != nil {
		log.Println("Failed to publish message:", err)
		return err
	}

	log.Println("Message published successfully:", message)
	return nil
}
```

```go
package kafka

import (
	"context"
	"github.com/segmentio/kafka-go"

func ConsumeMessages(topic string) {
	reader := kafka.NewReader(kafka.ReaderConfig{
		Brokers: []string{"localhost:9092"},
		Topic:   topic,
		GroupID: "my-group",
	})

	for {
		message, err := reader.ReadMessage(context.Background())

		}

		log.Printf("Received message: %s\n", string(message.Value))
	}
```

Функция `publishToKafka`:

```go
func publishToKafka(topic, message string) error {
	config := sarama.NewConfig()
	config.Producer.Return.Successes = true

	producer, err := sarama.NewSyncProducer([]string{"localhost:9092"}, config)
	if err != nil {
		return err
	}
	defer producer.Close()

	msg := &sarama.ProducerMessage{
		Topic: topic,
		Value: sarama.StringEncoder(message),
	}
	_, _, err = producer.SendMessage(msg)
	return err
}
```

### **3. Расширяем Kafka**

---

#### **3.1. Event Logger**

Этот сервис слушает события из Kafka и сохраняет их в PostgreSQL.

Пример:

```go
package main

import (
	"context"
	"log"

	"github.com/Shopify/sarama"
	"github.com/jackc/pgx/v5/pgxpool"
)

func main() {
	// Подключаемся к PostgreSQL
	db, err := pgxpool.New(context.Background(), "postgres://user:password@localhost:5432/app_db")
	if err != nil {
		log.Fatalf("Failed to connect to PostgreSQL: %v", err)
	}
	defer db.Close()

	// Создаём consumer для Kafka
	consumer, err := sarama.NewConsumer([]string{"localhost:9092"}, nil)
	if err != nil {
		log.Fatalf("Failed to start Kafka consumer: %v", err)
	}
	defer consumer.Close()

	partitionConsumer, err := consumer.ConsumePartition("user_events", 0, sarama.OffsetNewest)
	if err != nil {
		log.Fatalf("Failed to start Kafka partition consumer: %v", err)
	}
	defer partitionConsumer.Close()

	log.Println("Listening for Kafka events...")
	for msg := range partitionConsumer.Messages() {
		log.Printf("Received message: %s", string(msg.Value))

		// Сохраняем событие в PostgreSQL
		_, err := db.Exec(context.Background(), "INSERT INTO events (event_data) VALUES ($1)", string(msg.Value))
		if err != nil {
			log.Printf("Failed to save event: %v", err)
		}
	}
}
```

### **2.1. Транзакции в PostgreSQL**

Когда работаешь с критичными данными, транзакции помогают сохранить целостность.

#### **Пример транзакции в Go**

```go
func createUserWithTransaction(db *pgxpool.Pool, name, email string) error {
	tx, err := db.Begin(context.Background())
	if err != nil {
		return err
	}
	defer tx.Rollback(context.Background())

	// Вставка пользователя
	_, err = tx.Exec(context.Background(), "INSERT INTO users (name, email) VALUES ($1, $2)", name, email)
	if err != nil {
		return err
	}

	// Логирование события
	_, err = tx.Exec(context.Background(), "INSERT INTO events (event_data) VALUES ($1)", fmt.Sprintf("User %s created", name))
	if err != nil {
		return err
	}

	// Фиксируем транзакцию
	return tx.Commit(context.Background())
}
```

## **3. Kafka: ретраи, dead-letter-очереди и мониторинг**

---

### **3.1. Ретраи сообщений**

Для обработки ошибок при потреблении сообщений используем **Retry Mechanism**.

Пример потребления с ретраями:

```go
func consumeWithRetry(consumer sarama.Consumer, db *pgxpool.Pool) {
	partitionConsumer, _ := consumer.ConsumePartition("user_events", 0, sarama.OffsetNewest)
	defer partitionConsumer.Close()

	for msg := range partitionConsumer.Messages() {
		err := processMessage(db, msg)
		if err != nil {
			log.Printf("Error processing message: %v. Retrying...", err)
			go retryMessage(msg)
		}
	}
}

func retryMessage(msg *sarama.ConsumerMessage) {
	// Ретрай через задержку
	time.Sleep(5 * time.Second)
	log.Printf("Retrying message: %s", string(msg.Value))
}
```

---

### **3.2. Dead-letter Queue**

Если после нескольких ретраев сообщение не удалось обработать, отправляем его в отдельный топик.

Пример:

```go
func sendToDLQ(producer sarama.SyncProducer, msg *sarama.ConsumerMessage) {
	producer.SendMessage(&sarama.ProducerMessage{
		Topic: "dead_letter_queue",
		Value: sarama.StringEncoder(msg.Value),
	})
}
```

```yaml
  push:
    branches:
      - main

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
    - uses: actions/checkout@v3
    - name: Set up Go
      uses: actions/setup-go@v3
      with:
        go-version: 1.20
    - name: Install dependencies
      run: go mod tidy
    - name: Run tests
      run: go test ./...
    - name: Build Docker image
      run: docker build -t user-service:latest .
    - name: Push to Docker Hub
      run: docker push user-service:latest
```

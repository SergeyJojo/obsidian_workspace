## ✅ Цель задачи:
Написать программу, которая:
1. Последовательно делает HTTP-запросы по списку URL
2. Выводит `адрес url - ok` если статус 200, иначе `адрес url - not ok`
3. Переписать с использованием **горутин и каналов**, чтобы запросы шли параллельно
4. Убрать зависимость от длины списка (не использовать `len(urls)`)
5. Добавить тесты (можно моки)

---

## 🚀 Решение: по шагам

---

### 🔹 Шаг 1: Простая версия (последовательная)

```go
package main

import (
	"fmt"
	"net/http"
)

func main() {
	urls := []string{
		"http://ozon.ru",
		"https://ozon.ru",
		"http://google.com",
		"http://somesite.com",
		"http://non-existent.domain.tld",
		"https://ya.ru",
		"http://ya.ru",
		"http://ёёёё",
	}

	for _, url := range urls {
		resp, err := http.Get(url)
		if err != nil {
			fmt.Printf("%s - not ok\n", url)
			continue
		}
		defer resp.Body.Close()

		if resp.StatusCode == 200 {
			fmt.Printf("%s - ok\n", url)
		} else {
			fmt.Printf("%s - not ok\n", url)
		}
	}
}
```

> ⚠️ Это работает, но **последовательно** — медленно.

### 🔹 Шаг 2: Параллельная версия с горутинами и каналами

```go
package main

import (
	"fmt"
	"net/http"
	"sync"
)

type Result struct {
	URL     string
	Success bool
	Error   error
}

func checkURL(url string, ch chan<- Result) {
	resp, err := http.Get(url)
	if err != nil {
		ch <- Result{URL: url, Success: false, Error: err}
		return
	}
	defer resp.Body.Close()

	success := resp.StatusCode == 200
	ch <- Result{URL: url, Success: success, Error: nil}
}

func main() {
	urls := []string{
		"http://ozon.ru",
		"https://ozon.ru",
		"http://google.com",
		"http://somesite.com",
		"http://non-existent.domain.tld",
		"https://ya.ru",
		"http://ya.ru",
		"http://ёёёё",
	}

	// Канал для результатов
	results := make(chan Result, len(urls))
	var wg sync.WaitGroup

	// Запускаем горутины
	for _, url := range urls {
		wg.Add(1)
		go func(u string) {
			defer wg.Done()
			checkURL(u, results)
		}(url)
	}

	// Горутина для сбора результатов (в основном потоке)
	go func() {
		wg.Wait()
		close(results)
	}()

	// Печатаем в основном потоке
	for res := range results {
		if res.Success {
			fmt.Printf("%s - ok\n", res.URL)
		} else {
			fmt.Printf("%s - not ok\n", res.URL)
		}
	}
}
```

✅ Теперь:
- Запросы идут **параллельно**
- Печать происходит **в основном потоке**
- Не используется `len(urls)` — мы просто передаём каждый URL в горутину
- Используется `sync.WaitGroup` + канал для синхронизации

### 🔹 Шаг 3: Рефакторинг — вынос логики в функцию

```go
package main

import (
	"fmt"
	"net/http"
	"sync"
)

type Result struct {
	URL     string
	Success bool
	Error   error
}

func fetchURL(url string) Result {
	resp, err := http.Get(url)
	if err != nil {
		return Result{URL: url, Success: false, Error: err}
	}
	defer resp.Body.Close()

	return Result{URL: url, Success: resp.StatusCode == 200, Error: nil}
}

func processURLs(urls []string) <-chan Result {
	results := make(chan Result, len(urls))
	var wg sync.WaitGroup

	for _, url := range urls {
		wg.Add(1)
		go func(u string) {
			defer wg.Done()
			results <- fetchURL(u)
		}(url)
	}

	go func() {
		wg.Wait()
		close(results)
	}()

	return results
}

func main() {
	urls := []string{
		"http://ozon.ru",
		"https://ozon.ru",
		"http://google.com",
		"http://somesite.com",
		"http://non-existent.domain.tld",
		"https://ya.ru",
		"http://ya.ru",
		"http://ёёёё",
	}

	for res := range processURLs(urls) {
		if res.Success {
			fmt.Printf("%s - ok\n", res.URL)
		} else {
			fmt.Printf("%s - not ok\n", res.URL)
		}
	}
}
```

### 🔹 Шаг 4: Тесты (с моками)

Создадим интерфейс для `HTTPClient`, чтобы можно было подменять `http.Client`.

```go
// mock_test.go
package main

import (
	"errors"
	"net/http"
	"testing"
)

type MockHTTPClient struct {
	Responses map[string]*http.Response
	Errors    map[string]error
}

func (m *MockHTTPClient) Get(url string) (*http.Response, error) {
	if err, ok := m.Errors[url]; ok {
		return nil, err
	}
	if resp, ok := m.Responses[url]; ok {
		return resp, nil
	}
	return nil, errors.New("unknown URL")
}

func TestFetchURL(t *testing.T) {
	mockClient := &MockHTTPClient{
		Responses: map[string]*http.Response{
			"http://good.com": {StatusCode: 200},
		},
		Errors: map[string]error{
			"http://bad.com": errors.New("connection refused"),
		},
	}

	// Подменяем http.Get
	oldGet := httpGet
	httpGet = func(url string) (*http.Response, error) {
		return mockClient.Get(url)
	}
	defer func() { httpGet = oldGet }()

	result := fetchURL("http://good.com")
	if !result.Success {
		t.Error("Expected success for good.com")
	}

	result = fetchURL("http://bad.com")
	if result.Success {
		t.Error("Expected failure for bad.com")
	}
}

// Глобальная переменная для мокирования
var httpGet = http.Get
```

> 💡 Для полного тестирования нужно заменить `http.Get` — поэтому мы сделали `httpGet` как переменную.

### 🔹 Шаг 5: Ответ на вопрос "Какие тесты кандидат написал бы?"

Кандидат должен написать:

1. **Юнит-тесты для `fetchURL`** — с моками HTTP-клиента
2. **Тесты для `processURLs`** — проверка, что все URL обрабатываются
3. **Интеграционные тесты** — с реальными URL (например, `http://httpbin.org/status/200`)
4. **Тесты на обработку ошибок** — неправильный URL, таймаут, 404, 500

```go
// Kafka-продюсер для отправки событий о заказах

package kafka

import (
	"context"
	"encoding/json"
	"fmt"
	"order-service/internal/domain/order"

	"github.com/segmentio/kafka-go"
)

// OrderEventProducer публикует события заказов в Kafka

type OrderEventProducer struct {
	writer *kafka.Writer
	topic  string
}

func NewOrderEventProducer(brokers []string, topic string) *OrderEventProducer {
	writer := &kafka.Writer{
		Addr:     kafka.TCP(brokers...),
		Topic:    topic,
		Balancer: &kafka.LeastBytes{},
	}
	return &OrderEventProducer{writer: writer, topic: topic}
}

// Publish публикует событие "order_created" в Kafka
func (p *OrderEventProducer) Publish(ctx context.Context, o *order.Order) error {
	payload, err := json.Marshal(struct {
		Event string        `json:"event"`
		Data  *order.Order  `json:"data"`
	}{
		Event: "order_created",
		Data:  o,
	})
	if err != nil {
		return fmt.Errorf("marshal: %w", err)
	}

	msg := kafka.Message{
		Key:   []byte(o.ID),
		Value: payload,
	}
	return p.writer.WriteMessages(ctx, msg)
}
```

```go
// internal/app/app.go
// Инициализация инфраструктуры и запуск gRPC сервера с Kafka

package app

import (
	"context"
	"log"
	"net"
	"order-service/internal/infrastructure/kafka"
	"order-service/internal/infrastructure/postgres"
	"order-service/internal/transport/grpcserver"

	"github.com/jackc/pgx/v5/pgxpool"
	"google.golang.org/grpc"
)

func Run() error {
	ctx := context.Background()

	// Инициализация PostgreSQL
	pgxCfg, err := pgxpool.ParseConfig("postgres://user:pass@localhost:5432/orders_db?sslmode=disable")
	if err != nil {
		return err
	}
	pool, err := pgxpool.NewWithConfig(ctx, pgxCfg)
	if err != nil {
		return err
	}
	repo := postgres.NewOrderPGRepository(pool)

	// Инициализация Kafka producer
	kafkaProducer := kafka.NewOrderEventProducer([]string{"localhost:9092"}, "orders-events")

	// gRPC сервер
	grpcServer := grpc.NewServer()
	grpcserver.RegisterWithKafka(grpcServer, repo, kafkaProducer)

	ln, err := net.Listen("tcp", ":50051")
	if err != nil {
		return err
	}

	log.Println("Starting gRPC server on :50051")
	return grpcServer.Serve(ln)
}
```

```go
// internal/transport/grpcserver/server.go
// Расширение gRPC сервиса поддержкой Kafka

package grpcserver

import (
	"context"
	"order-service/internal/domain/order"
	kafkapub "order-service/internal/infrastructure/kafka"
	"order-service/internal/transport/grpcserver/pb"
	"time"

	"github.com/google/uuid"
	"google.golang.org/grpc"
	"google.golang.org/grpc/codes"
	"google.golang.org/grpc/status"
)

type OrderService struct {
	pb.UnimplementedOrderServiceServer
	repo   order.Repository
	kafka *kafkapub.OrderEventProducer
}

func NewOrderService(repo order.Repository, kafka *kafkapub.OrderEventProducer) *OrderService {
	return &OrderService{repo: repo, kafka: kafka}
}

func RegisterWithKafka(s *grpc.Server, repo order.Repository, kafka *kafkapub.OrderEventProducer) {
	pb.RegisterOrderServiceServer(s, NewOrderService(repo, kafka))
}

func (s *OrderService) HealthCheck(ctx context.Context, _ *pb.Empty) (*pb.HealthResponse, error) {
	return &pb.HealthResponse{Status: "ok"}, nil
}

func (s *OrderService) CreateOrder(ctx context.Context, req *pb.CreateOrderRequest) (*pb.CreateOrderResponse, error) {
	id := uuid.NewString()
	o := &order.Order{
		ID:        id,
		UserID:    req.User_id,
		Items:     req.Items,
		CreatedAt: time.Now(),
	}

	if err := s.repo.Create(ctx, o); err != nil {
		return nil, status.Errorf(codes.Internal, "db error: %v", err)
	}

	if err := s.kafka.Publish(ctx, o); err != nil {
		return nil, status.Errorf(codes.Internal, "kafka publish error: %v", err)
	}

	return &pb.CreateOrderResponse{Order_id: id}, nil
}

func (s *OrderService) GetOrder(ctx context.Context, req *pb.GetOrderRequest) (*pb.GetOrderResponse, error) {
	o, err := s.repo.Get(ctx, req.Order_id)
	if err != nil {
		return nil, status.Errorf(codes.NotFound, "order not found: %v", err)
	}

	return &pb.GetOrderResponse{
		Order_id:   o.ID,
		User_id:    o.UserID,
		Items:      o.Items,
		Created_at: o.CreatedAt.Format(time.RFC3339),
	}, nil
}
```

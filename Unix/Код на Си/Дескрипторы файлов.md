
Вот пример работы с файловыми дескрипторами на **C** в стиле Linux-документации, с подробными комментариями:

---

### 🔧 **Базовые операции с файловыми дескрипторами** (POSIX API)
```c
#include <fcntl.h>    // Для open(), O_* флагов
#include <unistd.h>   // Для read(), write(), close()
#include <stdio.h>    // Для perror()
#include <string.h>   // Для strlen()

int main() {
    int fd;  // Файловый дескриптор
    ssize_t bytes;  // Количество прочитанных/записанных байт

    // 1. Открытие файла (аналог open() в Linux)
    fd = open("example.txt", O_RDWR | O_CREAT, 0644);
    if (fd == -1) {
        perror("Ошибка открытия файла");
        return 1;
    }
    printf("Файл открыт, дескриптор: %d\n", fd);

    // 2. Запись в файл
    const char *text = "Hello, Linux descriptors!\n";
    bytes = write(fd, text, strlen(text));
    if (bytes == -1) {
        perror("Ошибка записи");
        close(fd);
        return 1;
    }
    printf("Записано %zd байт\n", bytes);

    // 3. Сдвиг указателя в начало (аналог lseek())
    off_t offset = lseek(fd, 0, SEEK_SET);
    if (offset == -1) {
        perror("Ошибка lseek");
        close(fd);
        return 1;
    }

    // 4. Чтение из файла
    char buffer[256];
    bytes = read(fd, buffer, sizeof(buffer) - 1);
    if (bytes == -1) {
        perror("Ошибка чтения");
        close(fd);
        return 1;
    }
    buffer[bytes] = '\0';  // Добавляем нуль-терминатор
    printf("Прочитано: %s\n", buffer);

    // 5. Закрытие дескриптора
    if (close(fd) == -1) {
        perror("Ошибка закрытия");
        return 1;
    }
    printf("Файл закрыт\n");

    return 0;
}
```

---

### 🛠 **Ключевые моменты из документации Linux**:
1. **Флаги открытия (`open()`)**:
   - `O_RDONLY` — только чтение  
   - `O_WRONLY` — только запись  
   - `O_RDWR` — чтение и запись  
   - `O_CREAT` — создать файл, если не существует (требует третий аргумент — режим доступа, например, `0644`)  
   - `O_APPEND` — запись в конец файла  
   - `O_TRUNC` — обрезать файл до нулевой длины  

2. **Ошибки**:
   - Все системные вызовы возвращают `-1` при ошибке.  
   - `errno` устанавливается в код ошибки (используйте `perror()` для вывода).  

3. **Важные системные вызовы**:
   - `read(fd, buf, count)` — возвращает количество прочитанных байт (`0` — конец файла).  
   - `write(fd, buf, count)` — возвращает количество записанных байт.  
   - `lseek(fd, offset, whence)` — перемещает указатель (`SEEK_SET`, `SEEK_CUR`, `SEEK_END`).

### ⚠ **Пример обработки ошибок (как в мануалах Linux)**:
```c
fd = open("/etc/passwd", O_RDONLY);
if (fd == -1) {
    if (errno == EACCES) {
        fprintf(stderr, "Доступ запрещен\n");
    } else if (errno == ENOENT) {
        fprintf(stderr, "Файл не существует\n");
    }
    return EXIT_FAILURE;
}
```

### 🔥 **Продвинутые возможности**:
1. **Неблокирующий I/O**:
   ```c
   fd = open("fifo", O_RDONLY | O_NONBLOCK);
   ```

2. **Дублирование дескрипторов**:
   ```c
   int new_fd = dup(fd);  // Создает копию дескриптора
   ```

3. **Файловые блокировки**:
   ```c
   struct flock fl = {
       .l_type = F_WRLCK,  // Тип блокировки (F_RDLCK, F_WRLCK, F_UNLCK)
       .l_whence = SEEK_SET,
       .l_start = 0,
       .l_len = 100  // Блокируем первые 100 байт
   };
   fcntl(fd, F_SETLK, &fl);  // Установка блокировки
   ```

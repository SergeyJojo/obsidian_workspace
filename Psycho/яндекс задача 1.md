Отличная задача! Реализуем `Balancer`, который будет распределять запросы между несколькими экземплярами `BackendImpl`.

Вот полное решение с комментариями:

```go
package main

import (
	"context"
	"math/rand"
	"sync"
	"time"
)

// Balancer реализует client-side балансировку нагрузки
type Balancer struct {
	backends []Backend       // список всех бэкендов
	mu       sync.RWMutex    // для потокобезопасности
}

// NewBalancer создает новый балансировщик
func NewBalancer(addrs []string) *Balancer {
	backends := make([]Backend, 0, len(addrs))
	
	// Создаем бэкенды для всех адресов
	for _, addr := range addrs {
		backend := NewBackend(addr)
		backends = append(backends, backend)
	}
	
	return &Balancer{
		backends: backends,
	}
}

// Invoke вызывает метод на одном из бэкендов
func (b *Balancer) Invoke(ctx context.Context, req Request) (Response, error) {
	// Получаем случайный бэкенд
	backend, err := b.getRandomBackend()
	if err != nil {
		return nil, err
	}
	
	// Вызываем метод на выбранном бэкенде
	return backend.Invoke(ctx, req)
}

// getRandomBackend возвращает случайный доступный бэкенд
func (b *Balancer) getRandomBackend() (Backend, error) {
	b.mu.RLock()
	defer b.mu.RUnlock()
	
	if len(b.backends) == 0 {
		return nil, ErrNoBackends
	}
	
	// Простой случайный выбор
	// В реальной реализации здесь должна быть проверка здоровья бэкендов
	index := rand.Intn(len(b.backends))
	return b.backends[index], nil
}

// UpdateBackends обновляет список бэкендов
func (b *Balancer) UpdateBackends(addrs []string) {
	b.mu.Lock()
	defer b.mu.Unlock()
	
	backends := make([]Backend, 0, len(addrs))
	for _, addr := range addrs {
		backend := NewBackend(addr)
		backends = append(backends, backend)
	}
	
	b.backends = backends
}

// ErrNoBackends ошибка когда нет доступных бэкендов
var ErrNoBackends = errors.New("no backends available")

func init() {
	rand.Seed(time.Now().UnixNano())
}
```

## 🚀 Улучшенная версия с проверкой здоровья бэкендов

Для production использования лучше добавить проверку здоровья:

```go
type Balancer struct {
	backends    []Backend
	mu          sync.RWMutex
	healthCheck *HealthChecker
}

type HealthChecker struct {
	interval   time.Duration
	timeout    time.Duration
	stopChan   chan struct{}
	isHealthy  func(Backend) bool
	healthyMap sync.Map // map[Backend]bool
}

// NewBalancerWithHealthCheck создает балансировщик с проверкой здоровья
func NewBalancerWithHealthCheck(addrs []string, checkInterval time.Duration) *Balancer {
	balancer := NewBalancer(addrs)
	balancer.healthCheck = NewHealthChecker(checkInterval, balancer.checkHealth)
	balancer.healthCheck.Start(balancer.backends)
	return balancer
}

// getHealthyBackend возвращает случайный здоровый бэкенд
func (b *Balancer) getHealthyBackend() (Backend, error) {
	b.mu.RLock()
	defer b.mu.RUnlock()
	
	var healthyBackends []Backend
	for _, backend := range b.backends {
		if b.healthCheck.IsHealthy(backend) {
			healthyBackends = append(healthyBackends, backend)
		}
	}
	
	if len(healthyBackends) == 0 {
		return nil, ErrNoHealthyBackends
	}
	
	index := rand.Intn(len(healthyBackends))
	return healthyBackends[index], nil
}

// checkHealth проверяет здоровье бэкенда
func (b *Balancer) checkHealth(backend Backend) bool {
	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()
	
	// Пытаемся сделать легковесный запрос
	_, err := backend.Invoke(ctx, &PingRequest{})
	return err == nil
}
```

## 📊 Альтернативные стратегии балансировки

Вместо случайного выбора можно реализовать другие стратегии:

```go
// RoundRobin балансировка
func (b *Balancer) getRoundRobinBackend() (Backend, error) {
	b.mu.RLock()
	defer b.mu.RUnlock()
	
	if len(b.backends) == 0 {
		return nil, ErrNoBackends
	}
	
	b.currentIndex = (b.currentIndex + 1) % len(b.backends)
	return b.backends[b.currentIndex], nil
}

// LeastConnections балансировка (наименьшее количество соединений)
func (b *Balancer) getLeastConnectionsBackend() (Backend, error) {
	b.mu.RLock()
	defer b.mu.RUnlock()
	
	if len(b.backends) == 0 {
		return nil, ErrNoBackends
	}
	
	var bestBackend Backend
	minConnections := int64(1<<63 - 1)
	
	for _, backend := range b.backends {
		if connCount := backend.GetConnectionCount(); connCount < minConnections {
			minConnections = connCount
			bestBackend = backend
		}
	}
	
	return bestBackend, nil
}
```

## 💡 Ключевые особенности реализации:

1. **Потокобезопасность** - используем `sync.RWMutex` для конкурентного доступа
2. **Простая стратегия** - случайный выбор для равномерного распределения
3. **Расширяемость** - легко добавить другие стратегии балансировки
4. **Инкапсуляция** - клиенты работают через тот же интерфейс `Backend`

Эта реализация обеспечивает базовую client-side балансировку нагрузки, которую можно расширять в зависимости от конкретных требований.

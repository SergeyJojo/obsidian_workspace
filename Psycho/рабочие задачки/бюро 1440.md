
// Описание:
// Нужно реализовать backend часть приложения планировщика, в котором будет 3 метода:

// 1) Создать команду с ID и временем запуска.
// 2) Получить список активных команд.
// 3) Отменить команду по ID.

// Команда должна выполнится точно в переданное в первом методе

// Методы будут вызываться через HTTP сервер, количество одновременных запросов
// этих методов не определно

// Само выполнение команды пускай будет Print(ID) в stdout, чтобы эмулировать выполнение.


Хорошая задача, Сергей 👍  
Давай сделаем её на Go — со всеми слоями: HTTP-сервер, хранение команд в памяти, конкурентная безопасная работа с планировщиком и точное выполнение в нужное время.

Основные моменты:

- Нужно хранить команды (`ID`, `время запуска`).
    
- Выполнять их строго во время (лучше через `time.AfterFunc` или отдельный планировщик-цикл).
    
- Обеспечить потокобезопасность при добавлении, отмене и получении списка (используем `sync.Mutex`).
    
- Для простоты возьмём хранение в памяти (map).
    

Вот минимальный, но рабочий пример:

```go
package main

import (
	"encoding/json"
	"fmt"
	"log"
	"net/http"
	"sync"
	"time"
)

type Command struct {
	ID    string    `json:"id"`
	RunAt time.Time `json:"run_at"`
	timer *time.Timer
}

type Scheduler struct {
	mu       sync.Mutex
	commands map[string]*Command
}

func NewScheduler() *Scheduler {
	return &Scheduler{
		commands: make(map[string]*Command),
	}
}

func (s *Scheduler) AddCommand(id string, runAt time.Time) error {
	s.mu.Lock()
	defer s.mu.Unlock()

	if _, exists := s.commands[id]; exists {
		return fmt.Errorf("command with id %s already exists", id)
	}

	delay := time.Until(runAt)
	if delay < 0 {
		return fmt.Errorf("time is in the past")
	}

	cmd := &Command{
		ID:    id,
		RunAt: runAt,
	}
	cmd.timer = time.AfterFunc(delay, func() {
		fmt.Println("Executing command:", id)
		s.mu.Lock()
		delete(s.commands, id) // после выполнения убираем
		s.mu.Unlock()
	})

	s.commands[id] = cmd
	return nil
}

func (s *Scheduler) ListCommands() []Command {
	s.mu.Lock()
	defer s.mu.Unlock()

	res := make([]Command, 0, len(s.commands))
	for _, c := range s.commands {
		res = append(res, *c)
	}
	return res
}

func (s *Scheduler) CancelCommand(id string) error {
	s.mu.Lock()
	defer s.mu.Unlock()

	cmd, ok := s.commands[id]
	if !ok {
		return fmt.Errorf("command not found")
	}
	if cmd.timer.Stop() {
		delete(s.commands, id)
		fmt.Println("Cancelled command:", id)
	}
	return nil
}

// HTTP handlers
func main() {
	scheduler := NewScheduler()

	http.HandleFunc("/create", func(w http.ResponseWriter, r *http.Request) {
		var req struct {
			ID    string `json:"id"`
			RunAt string `json:"run_at"`
		}
		if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
			http.Error(w, err.Error(), http.StatusBadRequest)
			return
		}
		t, err := time.Parse(time.RFC3339, req.RunAt)
		if err != nil {
			http.Error(w, "invalid time format, use RFC3339", http.StatusBadRequest)
			return
		}

		if err := scheduler.AddCommand(req.ID, t); err != nil {
			http.Error(w, err.Error(), http.StatusBadRequest)
			return
		}
		w.WriteHeader(http.StatusCreated)
	})

	http.HandleFunc("/list", func(w http.ResponseWriter, r *http.Request) {
		cmds := scheduler.ListCommands()
		_ = json.NewEncoder(w).Encode(cmds)
	})

	http.HandleFunc("/cancel", func(w http.ResponseWriter, r *http.Request) {
		var req struct {
			ID string `json:"id"`
		}
		if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
			http.Error(w, err.Error(), http.StatusBadRequest)
			return
		}
		if err := scheduler.CancelCommand(req.ID); err != nil {
			http.Error(w, err.Error(), http.StatusNotFound)
			return
		}
		w.WriteHeader(http.StatusOK)
	})

	fmt.Println("Server started at :8080")
	log.Fatal(http.ListenAndServe(":8080", nil))
}
```

Пример использования (через `curl`):

1. Создать задачу на будущее время:
    

```bash
curl -X POST http://localhost:8080/create \
  -d '{"id":"task1","run_at":"2025-09-04T15:10:00Z"}' \
  -H "Content-Type: application/json"
```

2. Список задач:
    

```bash
curl http://localhost:8080/list
```

3. Отменить:
    

```bash
curl -X POST http://localhost:8080/cancel \
  -d '{"id":"task1"}' \
  -H "Content-Type: application/json"
```

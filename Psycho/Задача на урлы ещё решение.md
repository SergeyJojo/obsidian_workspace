Тогда вот тебе готовый вариант под всё задание из скриншота — с поэтапной реализацией, но уже в финальной форме, где:

- URL-ы проверяются конкурентно.
    
- Количество одновременных горутин задаётся.
    
- Каналы используются для передачи результатов в основной поток.
    
- Работа прекращается после получения первых **2 успешных** ответов `200 OK`.
    

```go
package main

import (
	"fmt"
	"net/http"
	"time"
)

// структура для передачи результатов
type result struct {
	url    string
	status string
}

func main() {
	urls := []string{
		"http://ozon.ru",
		"https://ozon.ru",
		"http://google.com",
		"http://somesite.com",
		"http://non-existent.domain.tld",
		"https://ya.ru",
		"http://ya.ru",
		"http://ёёёё",
	}

	const maxWorkers = 3 // ограничение параллелизма
	const successLimit = 2

	results := make(chan result)
	done := make(chan struct{})

	// семафор для ограничения горутин
	sem := make(chan struct{}, maxWorkers)

	successCount := 0

	// запуск горутин
	for _, url := range urls {
		// проверка на досрочное завершение
		select {
		case <-done:
			break
		default:
		}

		sem <- struct{}{} // блокируем слот
		go func(u string) {
			defer func() { <-sem }()

			status := checkURL(u)
			results <- result{url: u, status: status}
		}(url)
	}

	// приём результатов в основном потоке
	go func() {
		for res := range results {
			fmt.Printf("%s — %s\n", res.url, res.status)
			if res.status == "ok" {
				successCount++
				if successCount >= successLimit {
					close(done)
					return
				}
			}
		}
	}()

	// ожидание завершения всех горутин
	for i := 0; i < cap(sem); i++ {
		sem <- struct{}{}
	}
	close(results)

	fmt.Println("Проверка завершена")
}

func checkURL(url string) string {
	client := http.Client{
		Timeout: 3 * time.Second,
	}

	resp, err := client.Get(url)
	if err != nil {
		return "not ok"
	}
	defer resp.Body.Close()

	if resp.StatusCode == http.StatusOK {
		return "ok"
	}
	return "not ok"
}
```

---

### Как это работает

1. **Семафор** (`sem`) ограничивает количество одновременных горутин (batched concurrency).
    
2. Все результаты складываются в **канал** `results`, который читает **только основной поток**.
    
3. Если набрано 2 успешных ответа — канал `done` закрывается, что останавливает дальнейшую отправку новых горутин.
    
4. `defer resp.Body.Close()` выполнится всегда, даже если запрос вернёт ошибку (если `resp != nil`), так что утечек нет.

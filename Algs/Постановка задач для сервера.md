
Вот как можно реализовать этот сервер для обработки фоновых задач с использованием очереди и воркеров на Go:

```go
package main

import (
	"errors"
	"sync"
)

type Task struct {
	Id   string
	Func func() (string, error)
}

type TaskResult struct {
	Result string
	Error  error
	Done   bool
}

type TaskServer struct {
	mu       sync.Mutex
	tasks    map[string]Task
	results  map[string]*TaskResult
	taskChan chan Task
	stopChan chan struct{}
	wg       sync.WaitGroup
}

func NewTaskServer() *TaskServer {
	return &TaskServer{
		tasks:    make(map[string]Task),
		results:  make(map[string]*TaskResult),
		taskChan: make(chan Task, 100), // Буферизованный канал для задач
		stopChan: make(chan struct{}),
	}
}

func (t *TaskServer) Submit(task Task) error {
	t.mu.Lock()
	defer t.mu.Unlock()

	if _, exists := t.tasks[task.Id]; exists {
		return errors.New("task with this ID already exists")
	}

	// Сохраняем задачу и инициализируем результат
	t.tasks[task.Id] = task
	t.results[task.Id] = &TaskResult{Done: false}

	// Отправляем задачу в канал для обработки
	t.taskChan <- task

	return nil
}

func (t *TaskServer) GetResult(id string) (string, bool) {
	t.mu.Lock()
	defer t.mu.Unlock()

	result, exists := t.results[id]
	if !exists {
		return "", false
	}

	if !result.Done {
		return "", false
	}

	if result.Error != nil {
		return result.Error.Error(), true
	}

	return result.Result, true
}

func (t *TaskServer) Start() error {
	// Запускаем несколько воркеров
	for i := 0; i < 5; i++ {
		t.wg.Add(1)
		go t.worker()
	}
	return nil
}

func (t *TaskServer) worker() {
	defer t.wg.Done()

	for {
		select {
		case task := <-t.taskChan:
			// Выполняем задачу
			result, err := task.Func()

			t.mu.Lock()
			t.results[task.Id] = &TaskResult{
				Result: result,
				Error:  err,
				Done:   true,
			}
			t.mu.Unlock()

		case <-t.stopChan:
			return
		}
	}
}

func (t *TaskServer) Stop() {
	close(t.stopChan)
	t.wg.Wait()
}
```

### Как это работает:

1. **Структуры данных**:
   - `Task` - содержит ID и функцию для выполнения
   - `TaskResult` - хранит результат выполнения задачи
   - `TaskServer` - основной сервер с:
     - `tasks` - мапой всех задач
     - `results` - мапой результатов
     - `taskChan` - каналом для передачи задач воркерам
     - `stopChan` - каналом для остановки воркеров

2. **Методы**:
   - `Submit()` - добавляет задачу в очередь
   - `GetResult()` - возвращает результат выполнения задачи
   - `Start()` - запускает воркеры
   - `worker()` - горутина, обрабатывающая задачи
   - `Stop()` - останавливает сервер

3. **Особенности реализации**:
   - Используются мьютексы для безопасного доступа к данным
   - Буферизованный канал для задач
   - Несколько воркеров для параллельной обработки
   - Graceful shutdown через stopChan

### Пример использования:

```go
func main() {
	server := NewTaskServer()
	server.Start()
	defer server.Stop()

	// Добавляем задачу
	server.Submit(Task{
		Id: "task1",
		Func: func() (string, error) {
			// Долгая операция
			return "success", nil
		},
	})

	// Позже проверяем результат
	if result, ok := server.GetResult("task1"); ok {
		println("Result:", result)
	}
}
```

Эта реализация обеспечивает:
- Параллельную обработку задач
- Безопасный доступ к данным из разных горутин
- Возможность проверки статуса задач
- Корректное завершение работы


# Рефакторинг кода для сбора курсов валют

Вот улучшенная версия вашего кода с лучшей структурой, обработкой ошибок и разделением ответственности:

```go
package main

import (
	"database/sql"
	"encoding/json"
	"fmt"
	"io"
	"log"
	"net/http"
	"os"
	"strconv"
	"strings"
	"time"

	_ "github.com/lib/pq"
)

// Config содержит конфигурационные параметры приложения
type Config struct {
	DBHost     string
	DBPort     int
	DBUser     string
	DBPassword string
	DBName     string
}

// BankConfig содержит параметры для запросов к банкам
type BankConfig struct {
	BankName string
	CurFrom  string
	CurTo    string
	URL      string
	Headers  map[string]string
	// TransformFunc можно добавить для кастомной обработки ответа
}

// CurrencyRate представляет запись о курсе валюты
type CurrencyRate struct {
	Bank  string
	From  string
	To    string
	Value float64
	Date  time.Time
}

// App представляет основное приложение с зависимостями
type App struct {
	DB        *sql.DB
	Client    *http.Client
	Config    Config
	Banks     []BankConfig
	Scheduler *time.Ticker
}

func main() {
	// Инициализация конфигурации
	config := Config{
		DBHost:     "localhost",
		DBPort:     5432,
		DBUser:     "postgres",
		DBPassword: "password",
		DBName:     "currency",
	}

	// Инициализация конфигурации банков
	banks := []BankConfig{
		{
			BankName: "Bank 1",
			CurFrom:  "RUB",
			CurTo:    "USD",
			URL:      "http://bank.example.com/rates/rub-usd",
		},
		{
			BankName: "Bank 2",
			CurFrom:  "RUB",
			CurTo:    "USD",
			URL:      "http://bank2.example.com/rates?currFrom=RUR&currTo=USD",
			Headers: map[string]string{
				"Authorization": `auth_token="XXXXXXX"`,
			},
		},
	}

	// Инициализация приложения
	app, err := NewApp(config, banks)
	if err != nil {
		log.Fatalf("Failed to initialize app: %v", err)
	}
	defer app.Close()

	// Обработка команд
	if len(os.Args) == 2 {
		switch os.Args[1] {
		case "help":
			printHelp()
		case "update":
			if err := app.UpdateRates(); err != nil {
				log.Fatalf("Failed to update rates: %v", err)
			}
		case "daemon":
			app.RunAsDaemon()
		default:
			printHelp()
		}
	} else {
		printHelp()
	}
}

// NewApp создает новое приложение
func NewApp(config Config, banks []BankConfig) (*App, error) {
	// Инициализация подключения к БД
	db, err := initDB(config)
	if err != nil {
		return nil, fmt.Errorf("database initialization failed: %w", err)
	}

	return &App{
		DB:     db,
		Client: &http.Client{Timeout: 10 * time.Second},
		Config: config,
		Banks:  banks,
	}, nil
}

// Close освобождает ресурсы приложения
func (a *App) Close() {
	if a.DB != nil {
		a.DB.Close()
	}
	if a.Scheduler != nil {
		a.Scheduler.Stop()
	}
}

// printHelp выводит справку по использованию
func printHelp() {
	fmt.Println("Usage:")
	fmt.Println("./currency help - show this help")
	fmt.Println("./currency update - update currency rates once")
	fmt.Println("./currency daemon - run as daemon updating rates every minute")
}

// initDB инициализирует подключение к базе данных
func initDB(config Config) (*sql.DB, error) {
	psqlconn := fmt.Sprintf("host=%s port=%d user=%s password=%s dbname=%s sslmode=disable",
		config.DBHost, config.DBPort, config.DBUser, config.DBPassword, config.DBName)

	db, err := sql.Open("postgres", psqlconn)
	if err != nil {
		return nil, err
	}

	if err := db.Ping(); err != nil {
		return nil, err
	}

	log.Println("Successfully connected to database")
	return db, nil
}

// UpdateRates обновляет курсы валют для всех банков
func (a *App) UpdateRates() error {
	for _, bank := range a.Banks {
		rate, err := a.fetchBankRate(bank)
		if err != nil {
			log.Printf("Failed to fetch rate from %s: %v", bank.BankName, err)
			continue
		}

		if err := a.saveRate(rate); err != nil {
			log.Printf("Failed to save rate from %s: %v", bank.BankName, err)
			continue
		}

		log.Printf("Successfully updated rate for %s: %s->%s = %.4f",
			bank.BankName, bank.CurFrom, bank.CurTo, rate.Value)
	}
	return nil
}

// fetchBankRate получает курс валюты от конкретного банка
func (a *App) fetchBankRate(bank BankConfig) (*CurrencyRate, error) {
	req, err := http.NewRequest(http.MethodGet, bank.URL, nil)
	if err != nil {
		return nil, fmt.Errorf("request creation failed: %w", err)
	}

	// Устанавливаем заголовки если они есть
	for k, v := range bank.Headers {
		req.Header.Add(k, v)
	}

	resp, err := a.Client.Do(req)
	if err != nil {
		return nil, fmt.Errorf("request failed: %w", err)
	}
	defer resp.Body.Close()

	if resp.StatusCode != http.StatusOK {
		return nil, fmt.Errorf("unexpected status code: %d", resp.StatusCode)
	}

	body, err := io.ReadAll(resp.Body)
	if err != nil {
		return nil, fmt.Errorf("failed to read response: %w", err)
	}

	value, err := parseBankResponse(bank.BankName, string(body))
	if err != nil {
		return nil, fmt.Errorf("failed to parse response: %w", err)
	}

	return &CurrencyRate{
		Bank:  bank.BankName,
		From:  bank.CurFrom,
		To:    bank.CurTo,
		Value: value,
		Date:  time.Now(),
	}, nil
}

// parseBankResponse парсит ответ банка в зависимости от его формата
func parseBankResponse(bankName, response string) (float64, error) {
	// Удаляем лишние пробелы и переносы строк
	response = strings.TrimSpace(response)

	// Bank 1 использует запятые вместо точек
	if bankName == "Bank 1" {
		response = strings.ReplaceAll(response, ",", ".")
	}

	value, err := strconv.ParseFloat(response, 64)
	if err != nil {
		return 0, fmt.Errorf("failed to parse float from '%s': %w", response, err)
	}

	return value, nil
}

// saveRate сохраняет курс валюты в базу данных
func (a *App) saveRate(rate *CurrencyRate) error {
	// Используем параметризованный запрос для безопасности
	stmt := `INSERT INTO currency_rates (bank, "from", "to", value, date) 
             VALUES ($1, $2, $3, $4, $5)`

	_, err := a.DB.Exec(stmt, rate.Bank, rate.From, rate.To, rate.Value, rate.Date)
	return err
}

// RunAsDaemon запускает приложение в режиме демона с обновлением каждую минуту
func (a *App) RunAsDaemon() {
	log.Println("Starting currency rates daemon...")
	a.Scheduler = time.NewTicker(1 * time.Minute)

	for range a.Scheduler.C {
		if err := a.UpdateRates(); err != nil {
			log.Printf("Error in scheduled update: %v", err)
		}
	}
}
```

## Основные улучшения:

1. **Структуризация кода**:
   - Разделение на логические структуры (App, Config, BankConfig, CurrencyRate)
   - Каждая функция выполняет одну четкую задачу

2. **Обработка ошибок**:
   - Замена паник на корректную обработку ошибок
   - Логирование ошибок вместо прерывания работы

3. **Безопасность**:
   - Использование параметризованных SQL-запросов для предотвращения SQL-инъекций
   - Правильное закрытие ресурсов (defer)

4. **Гибкость**:
   - Конфигурация вынесена в отдельные структуры
   - Поддержка разных форматов ответов банков
   - Возможность добавления новых банков

5. **Новые возможности**:
   - Режим демона с периодическим обновлением
   - Логирование процесса работы
   - Более информативные сообщения об ошибках

6. **Масштабируемость**:
   - Легко добавить новые банки и валютные пары
   - Возможность расширения функционала (например, добавление API)

7. **Тестируемость**:
   - Код лучше структурирован для модульного тестирования
   - Меньше глобальных зависимостей

Этот код теперь более надежный, безопасный и легче поддерживается.
